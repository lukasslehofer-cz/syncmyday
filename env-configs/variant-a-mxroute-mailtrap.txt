# ============================================
# VARIANTA A: MXroute + Mailtrap Setup
# ============================================
# MXroute: Systémové emaily (info@) - kritické, nízký objem
# Mailtrap: Blocker emaily (events@) - high volume
# ============================================

# ============================================
# MXroute SMTP (pro info@ systémové emaily)
# ============================================
# Použití: Welcome, Trial, Payment, Password Reset
# Objem: Nízký (5,000-20,000/rok)
# ============================================
MXROUTE_HOST=bunny.mxroute.com
MXROUTE_PORT=587
MXROUTE_USERNAME=info@syncmyday.cz
MXROUTE_PASSWORD=your_mxroute_password_here
MXROUTE_ENCRYPTION=tls

# ============================================
# Mailtrap SMTP (pro events@ blocker emaily)
# ============================================
# Použití: Calendar blocker invitations (iMIP)
# Objem: Vysoký (30K-150K/měsíc)
# ============================================
MAILTRAP_HOST=live.smtp.mailtrap.io
MAILTRAP_PORT=587
MAILTRAP_USERNAME=api
MAILTRAP_PASSWORD=your_mailtrap_api_token_here
MAILTRAP_ENCRYPTION=tls

# ============================================
# Mailtrap Inbound (příchozí kalendářové emaily)
# ============================================
# Webhook endpoint: /api/webhook/mailtrap-inbound
# Auto-cleanup: 7 days
# ============================================
MAILTRAP_INBOUND_SECRET=your_webhook_secret_here
MAILTRAP_INBOUND_EMAIL=abc123@inbound.mailtrap.io

# ============================================
# Mail FROM name (stejné pro všechny)
# ============================================
MAIL_FROM_NAME="SyncMyDay"

# ============================================
# Legacy/Fallback (pro backward compatibility)
# ============================================
MAIL_MAILER=smtp
MAIL_HOST=bunny.mxroute.com
MAIL_PORT=587
MAIL_USERNAME=info@syncmyday.cz
MAIL_PASSWORD=your_mxroute_password_here
MAIL_ENCRYPTION=tls

# ============================================
# JAK TO NASTAVIT:
# ============================================

# 1. MXroute DirectAdmin:
#    - Přidat 5 domén (cz, sk, pl, de, eu)
#    - Vytvořit info@ mailbox pro každou doménu
#    - Zkopírovat SMTP credentials
#    - Zkopírovat DKIM klíče pro DNS

# 2. Mailtrap Dashboard:
#    - Vytvořit účet (Essential plan: $15/měsíc)
#    - Přidat 5 domén do Sending Domains
#    - Verify každou doménu (DNS records)
#    - Zkopírovat SMTP API token
#    - Vytvořit Inbound inbox
#    - Nastavit webhook na /api/webhook/mailtrap-inbound

# 3. DNS pro KAŽDOU doménu:
#    @ MX 10 bunny.mxroute.com.
#    @ TXT "v=spf1 include:mxroute.com include:mailtrap.io ~all"
#    default._domainkey TXT "v=DKIM1; k=rsa; p=..." (z MXroute)
#    mtdkim._domainkey TXT "v=DKIM1; k=rsa; p=..." (z Mailtrap)
#    _dmarc TXT "v=DMARC1; p=quarantine; rua=mailto:dmarc@syncmyday.cz"

# 4. Deploy:
#    php artisan migrate
#    php artisan config:clear && php artisan config:cache

# ============================================
# JAK TO FUNGUJE:
# ============================================

# Systémové emaily (info@):
# User registrace → User.registration_domain = "syncmyday.pl"
# Welcome email → EmailHelper vrátí mailer='mxroute', address='info@syncmyday.pl'
# Email se pošle přes MXroute SMTP

# Blocker emaily (events@):
# Sync rule vytvoří blocker → EmailHelper vrátí mailer='mailtrap', address='events@syncmyday.pl'
# Email se pošle přes Mailtrap SMTP
# Mailtrap handleuje burst sending a rate limiting

# Příchozí emaily:
# Forward na abc123@inbound.mailtrap.io
# Mailtrap webhook zavolá /api/webhook/mailtrap-inbound
# Parser zpracuje .ics a vytvoří blockery
# Mailtrap automaticky maže staré emaily po 7 dnech
